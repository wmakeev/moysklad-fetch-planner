const inferred = new OperationPlanner(globalThis.fetch)
type NativeInit = Init<typeof globalThis.fetch>
export function exercise<T>(
  planner: Planner<NativeInit, Response>,
  decode: (bytes: Uint8Array) => Promise<T>
) {
  const handle = planner.submit({
    method: 'PUT',
    entityKey: 'entity-1',
    replay: { serverSafe: false, consumeSafe: false },
    prepare(ctx) {
      ctx.onDispose(async () => {
        await Promise.resolve()
      })
      const signal: AbortSignal = ctx.signal
      void signal
      return {
        url: new URL('https://offline.invalid'),
        init: { body: new ReadableStream<Uint8Array>(), duplex: 'half' }
      }
    },
    async consume(response, ctx) {
      ctx.onDiscard(() => undefined)
      if (response.status === 503) throw ctx.retryableHttpFailure(new Error('whole-operation failure'))
      try {
        return await decode(await response.bytes())
      } catch (cause) {
        throw ctx.retryableReadFailure(cause)
      }
    }
  })
  const typed: Handle<T> = handle
  const result: Promise<Completed<T>> = typed.result
  const observed = typed.observe(event => {
    const seq: number = event.sequence
    void seq
  })
  const phase: string = observed.snapshot.phase
  observed.unsubscribe()
  void result
  void phase
  return typed
}

  }
}
