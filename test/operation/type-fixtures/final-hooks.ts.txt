import { fetch } from 'undici'
import { FetchPlanner as RootPlanner } from '../../src/index.js'
import { FetchPlanner as DeepPlanner } from '../../src/planner.js'
import type { Spec, HookContext, Decision, Init, Handle, Hook } from '../../src/index.js'
import type { HookContext as DeepContext, Decision as DeepDecision, Hook as DeepHook } from '../../src/operation-types.js'
export const spec: Spec<Init<typeof globalThis.fetch>, Response, number> = {
  method: 'GET',
  replay: { serverSafe: true, consumeSafe: true },
  prepare: () => ({ url: '', init: { method: undefined, signal: undefined } }),
  consume: async response => (await response.bytes()).length,
  beforeDispatch(ctx) {
    const signal: AbortSignal = ctx.signal
    const sequence: number = ctx.snapshot.sequence
    void signal
    void sequence
    return { type: 'allow' }
  },
  async beforeRetry(ctx) {
    await Promise.resolve()
    const status: number | undefined = ctx.status
    void status
    return { type: 'defer', delayMs: 1 }
  }
}
export function deny(ctx: HookContext): Decision {
  const attempt: number = ctx.attemptNumber
  void attempt
  return { type: 'deny', code: 'POLICY_DENIED' }
}

const hook: Hook = deny
const deepHook: DeepHook = hook
const deepDecision: DeepDecision = { type: 'defer', delayMs: 1 }
export function deepDeny(context: DeepContext): DeepDecision {
  return deny(context)
}
void deepDecision
for (const Planner of [RootPlanner, DeepPlanner]) {
  const native = new Planner(globalThis.fetch)
  const nativeResult: Handle<number> = native.submit(spec)
  const undici = new Planner(fetch)
  const undiciResult: Handle<number> = undici.submit({
    method: 'POST',
    replay: { serverSafe: false, consumeSafe: false },
    prepare: () => ({ url: '', init: { body: 'payload' } }),
    consume: async response => (await response.arrayBuffer()).byteLength,
    beforeDispatch: deepHook,
    beforeRetry: async context => ({ type: 'defer', delayMs: context.attemptNumber })
  })
  void nativeResult
  void undiciResult
}
