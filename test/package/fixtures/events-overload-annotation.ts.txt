import { FetchPlanner } from '../../../src/index.js'
import type { HookContext } from '../../../src/index.js'
import type { fetch } from 'undici-types'
declare const transport: typeof fetch
const planner = new FetchPlanner(transport)
const control = (context: HookContext) => ({ type: 'allow' as const, status: context.status })
planner.observe(control)
planner.submit({
  method: 'GET',
  replay: { serverSafe: true, consumeSafe: true },
  prepare: () => ({ url: 'https://local.invalid/', init: {} }),
  consume: (response: Response) => response.bytes(),
  beforeDispatch: 42
})
