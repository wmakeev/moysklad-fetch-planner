## Отрицательные coverage-пробы: полный код

Каждая мутация применяется к копии подлинного LCOV в scratch, исходный coverage/lcov.info не меняется. В wrapFetch-zero добавлен исполненный посторонний function counter, чтобы проверить именно обязательность wrapFetch после общего function-gate.

```python
from pathlib import Path
import subprocess, re

scratch = Path('/tmp/orchestrate-fetch-planner-003/T70-1')
report = Path('coverage/lcov.info').read_text()
records = [r.strip()+'\nend_of_record\n' for r in report.split('end_of_record') if r.strip()]
modules = ['index', 'planner', 'pacing', 'response-policy', 'runtime']

def source(record):
    return re.search(r'^SF:(.+)$', record, re.M).group(1)

def run(name, text, success=False):
    path = scratch / (name+'.lcov')
    path.write_text(text)
    cmd = ['node', 'scripts/check-coverage.mjs', str(path)]
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=15)
    print('$ '+' '.join(cmd))
    print(result.stdout+result.stderr, end='')
    print('exit='+str(result.returncode))
    assert (result.returncode==0)==success, name
    print('PASS '+name)

run('valid', report, success=True)
for module in modules:
    target = 'src/'+module+'.ts'
    assert any(source(r)==target for r in records), target
    run(module+'-missing', ''.join(r for r in records if source(r)!=target))
    for kind in ['zero', 'zero-lines', 'zero-functions']:
        mutated=[]
        for record in records:
            if source(record)==target:
                if kind!='zero-functions':
                    record=re.sub(r'^(DA:\d+,)\d+', r'\g<1>0', record, flags=re.M)
                if kind!='zero-lines':
                    record=re.sub(r'^(FNDA:)\d+', r'\g<1>0', record, flags=re.M)
            mutated.append(record)
        run(module+'-'+kind, ''.join(mutated))
run('empty', '')
run('index-only', ''.join(r for r in records if source(r)=='src/index.ts'))
for foreign in ['test/index.test.ts', 'samples/demo.ts']:
    run(foreign.split('/')[0]+'-only', re.sub(r'^SF:.+$', 'SF:'+foreign, report, flags=re.M))
run('wrapFetch-zero', re.sub(r'^FNDA:\d+,wrapFetch$', 'FNDA:0,wrapFetch\nFNDA:1,anotherFunction', report, flags=re.M))
print('PASS: valid report accepted; 25 negative reports rejected, including missing/zero/zero-lines/zero-functions for every executable module.')
```

Все enqueue/retry/settle/release остаются в том же классе. pump остаётся единственной точкой перепланирования; scheduleWake хранит один timeout с проверкой владения. EventEmitter получает тот же экземпляр subclass, а публичный constructor импортируется через re-export без промежуточного экземпляра или wrapper-класса.
