## Краевые пути и вывод

- max1/2/4: repeated overflow до насыщения и после него; correction после каждого перехода в [1-max,0], effective в [1,max], минимум равен 1. Собственная проба дополнительно проверяет непосредственно stored parallelLimitCorrection, а не только getter [VERIFY].
- Несколько уже отправленных attempts при max2/4, late success и повторный overflow на минимуме: существующие transport сохраняют inflight до settlement; enqueue и reentrant enqueue не запускаются сверх уменьшенного лимита. Самостоятельная проба отвечает overflow на все изначальные inflight и подтверждает сохранение pending attempts до каждого ответа.
- Slot priority: меньший priority допускается первым, enqueue из slot notification не обходит effective limit; protected subclass hook действительно вызывается. Очередь и inflight в конце равны нулю.
- Recovery: max1/2/4 × random0/0.5/1, jitter0.3, прежняя формула period=1000*(1+2*random*0.3), проверка границы period и period+1, последовательное восстановление до max и отсутствие положительной correction после дополнительных периодов; admission после восстановления снова достигает max.
- Независимые Date jumps в обе стороны не двигают performance clock и не восстанавливают correction; монотонный период восстанавливает. Собственная смешанная проба использует скачки +1e9 и -2e9, одновременно с очередью и действующей серверной паузой.
- ServerNotBefore: самостоятельная проба после снижения создаёт отдельный RATE_LIMIT_OVERFLOW с Retry-After400; ни enqueue, ни Date jumps, ни граница399 мс не запускают retry, на400 мс стартует ровно один. Все scheduler T40 сценарии также проходят в check/coverage, включая late headers, max deadlines, pacing и bounded timers.
- Сохраняемые lifecycle/observer policy/pacing/runtime suites T10–T40 включены в общий test entrypoint и проходят. Diff реализации меняет только reduceParallelLimit, package.json лишь добавляет suite; test/index.test.ts и принятые runtime helpers не менялись. errorLimits и фасад не затронуты. Новых defaults/API/политик не вводится [SURFACE].
- Red/green: в копии временно заменён только src/index.ts на git show HEAD:src/index.ts. Старый код успешно компилируется, затем падают именно correction assertions: для max2 сохранено +1 вместо -1, для max4 +3 вместо -3. Новая suite даёт 10 assertion failures; собственная проба даёт два. max1 корректен и в baseline, поэтому его green до T50 ожидаем. После возврата исправления обе проверки дают 17/17.

## independent.mjs

```js
import assert from 'node:assert/strict'
import { test } from 'node:test'
import { Response } from 'undici'
import { FetchPlanner } from './__temp/test-build/src/index.js'
import { createRuntime } from './__temp/test-build/test/helpers/runtime.js'

for (const max of [1, 2, 4]) test(`Independent mixed overflow, pause, clock and slot admission max ${max}`, async t => {
  const runtime = createRuntime(t)
  let starts = 0
  const pending = []
  class Planner extends FetchPlanner {
    checks = 0
    checkFreeRequestSlotTrigger() { this.checks++; super.checkFreeRequestSlotTrigger() }
  }
  const planner = new Planner(() => {
    starts++
    return new Promise(resolve => pending.push(resolve))
  }, { maxParallelLimit: max, jitter: 0, parallelLimitCorrectionPeriodMs: 1000 })
  const fetch = planner.getFetch()
  const results = Array.from({length:max}, (_, i) => fetch(`http://offline.test/${i}`))
  const invariant = expected => {
    assert.equal(planner.getParallelLimitCorrection(), expected)
    assert.equal(planner.parallelLimitCorrection, expected, 'Stored state itself is bounded')
    assert.ok(expected >= 1-max && expected <= 0)
    assert.ok(max+expected >= 1 && max+expected <= max)
  }
  const reply = (status=200, pause) => {
    assert.ok(pending.length)
    const headers = status===429 ? {'X-Lognex-Auth':'429005'} : {}
    if(pause!==undefined) headers['X-Lognex-Retry-After']=String(pause)
    pending.shift()(new Response(null,{status,headers}))
  }
  await runtime.advance(0)
  assert.equal(starts,max)
  for(let i=0; i<max; i++) {
    reply(429,400)
    await runtime.flush()
    invariant(Math.max(1-max,-i-1))
    assert.equal(starts,max,'No replacement while original attempts remain or pause holds')
    assert.equal(pending.length,max-i-1,'Sent attempts remain owned until their replies')
  }
  await runtime.advance(0)
  assert.equal(pending.length,1)
  pending.shift()(new Response(null,{status:429,headers:{'X-Lognex-Retry-After':'400'}}))
  await runtime.flush()
  const pausedStarts=starts
  results.push(fetch('http://offline.test/enqueued'))
  const order=[]
  const high=planner.waitForFreeRequestSlot(-3).then(()=>{order.push(-3); results.push(fetch('http://offline.test/high'))})
  const low=planner.waitForFreeRequestSlot(3).then(()=>{order.push(3); results.push(fetch('http://offline.test/low'))})
  for(const jump of [1e9,-2e9]) {
    runtime.jumpWall(jump)
    await runtime.advance(0)
    invariant(1-max)
    assert.equal(starts,pausedStarts)
  }
  await runtime.advance(399)
  assert.equal(starts,pausedStarts)
  await runtime.advance(1)
  assert.equal(starts,pausedStarts+1)
  reply(429)
  await runtime.advance(0)
  invariant(1-max)
  assert.equal(pending.length,1)
  for(let i=0;i<max+3;i++) {
    assert.equal(pending.length,1)
    reply()
    await runtime.advance(0)
    invariant(1-max)
  }
  await Promise.all([high,low,...results])
  assert.deepEqual(order,[-3,3])
  assert.equal(planner.getCurInflightRequestsCount(),0)
  assert.equal(planner.getActionsQueueLength(),0)
  assert.ok(planner.checks>max)
  for(let expected=2-max; expected<=0; expected++) {
    await runtime.advance(1001)
    const result=fetch('http://offline.test/recover')
    await runtime.advance(0)
    invariant(expected)
    reply()
    await runtime.flush()
    await result
  }
  console.log(`Independent max ${max}: stored correction, inflight, pause, Date jumps, priorities and recovery verified`)
})
```
