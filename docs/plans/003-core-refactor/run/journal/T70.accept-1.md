```python
from pathlib import Path
import subprocess,re,json
w=Path.cwd(); records=[s.strip()+'\nend_of_record\n' for s in (w/'coverage/lcov.info').read_text().split('end_of_record') if s.strip()]
def source(s): return re.search(r'^SF:(.*)$',s,re.M)[1]
def zeros(s):
 s=re.sub(r'^(DA:\d+,)\d+',r'\g<1>0',s,flags=re.M)
 return re.sub(r'^FNDA:\d+,', 'FNDA:0,',s,flags=re.M)
variants={'valid':(''.join(records),0),'empty':('',1),'index-only':(''.join(r for r in records if source(r)=='src/index.ts'),1),'all-zero':(''.join(map(zeros,records)),1),'new-modules-zero':(''.join(r if source(r)=='src/index.ts' else zeros(r) for r in records),1)}
for name in ['index','planner','pacing','response-policy','runtime']:
 sf=f'src/{name}.ts'
 variants[f'{name}-missing']=(''.join(r for r in records if source(r)!=sf),1)
 variants[f'{name}-zero']=(''.join(zeros(r) if source(r)==sf else r for r in records),1)
 variants[f'{name}-lines-zero']=(''.join(re.sub(r'^(DA:\d+,)\d+',r'\g<1>0',r,flags=re.M) if source(r)==sf else r for r in records),1)
 variants[f'{name}-functions-zero']=(''.join(re.sub(r'^FNDA:\d+,','FNDA:0,',r,flags=re.M) if source(r)==sf else r for r in records),1)
for folder in ['test','samples','outside/src','src/../test']:
 variants[folder.replace('/','-')+'-impostor']=(''.join(re.sub(r'^SF:src/', 'SF:'+folder+'/',r,flags=re.M) for r in records),1)
```

### Собственная compatibility-проба: полный код

```typescript
import assert from 'node:assert/strict'
import { EventEmitter } from 'node:events'
import { test } from 'node:test'
import { Response } from 'undici'
import { FetchPlanner, wrapFetch, type FetchPlannerEventMap } from '../src/index.js'
import { createRuntime } from './helpers/runtime.js'

class AcceptancePlanner extends FetchPlanner {
  checks = 0
  protected override checkFreeRequestSlotTrigger() {
    this.checks++
    super.checkFreeRequestSlotTrigger()
  }
  snapshot() { return [this.retryTimeInterval, this.rateLimit, this.rateLimitRemaining] }
}

test('Independent subclass preserves FIFO retry, slot priority, events and getters', async t => {
  const runtime = createRuntime(t)
  const emitter = new EventEmitter<FetchPlannerEventMap>()
  const starts: string[] = []
  const replies = Array.from({ length: 3 }, () => {
    let resolve!: (value: Response) => void
    const promise = new Promise<Response>(done => { resolve = done })
    return { resolve, promise }
  })
  const payload = { method: 'POST', body: 'unchanged' }
  const planner = new AcceptancePlanner((url, init) => {
    starts.push(String(url))
    assert.equal(init, payload)
    return replies[starts.length - 1]!.promise
  }, { maxParallelLimit: 1, jitter: 0, eventHandler: emitter })
  const events: string[] = []
  emitter.on('request', (ev, instance) => { assert.equal(instance, planner); events.push(`request:${ev.actionId}:${ev.requestId}`) })
  emitter.on('response', (ev, instance) => { assert.equal(instance, planner); events.push(`response:${ev.actionId}:${ev.responseType}`) })
  const fetch = planner.getFetch()
  const first = fetch('http://offline.test/first', payload)
  const second = fetch('http://offline.test/second', payload)
  const slots: number[] = []
  const waits = [10, -5, 0].map(priority => planner.waitForFreeRequestSlot(priority).then(() => { slots.push(priority) }))
  assert.equal(planner.getActionsQueueLength(), 2)
  assert.equal(planner.getRequestSlotHandlersCount(), 3)
  await runtime.advance(0)
  assert.equal(planner.getCurInflightRequestsCount(), 1)
  replies[0]!.resolve(new Response(null, { status: 429, headers: { 'X-Lognex-Retry-After': '40' } }))
  await runtime.flush()
  assert.equal(planner.getCurInflightRequestsCount(), 0)
  assert.equal(planner.getActionsQueueLength(), 2)
  assert.equal(planner.getNextRequestTime(), Date.now() + 40)
  await runtime.advance(39)
  assert.deepEqual(starts, ['http://offline.test/first'])
  await runtime.advance(1)
  assert.deepEqual(starts, ['http://offline.test/first', 'http://offline.test/first'])
  const one = new Response('first', { status: 500 })
  replies[1]!.resolve(one)
  await runtime.flush()
  assert.equal(await first, one)
  await runtime.advance(0)
  assert.deepEqual(starts, ['http://offline.test/first', 'http://offline.test/first', 'http://offline.test/second'])
  const two = new Response('second')
  replies[2]!.resolve(two)
  await runtime.flush()
  assert.equal(await second, two)
  for (let i = 0; i < 4; i++) await runtime.flush()
  await Promise.all(waits)
  assert.deepEqual(slots, [-5, 0, 10])
  assert.deepEqual(events, ['request:1:1', 'response:1:RATE_LIMIT_OVERFLOW', 'request:1:2', 'response:1:OK', 'request:2:3', 'response:2:OK'])
  assert.deepEqual(planner.snapshot(), [3000, null, null])
  assert.equal(planner.getNextRequestTime(), 0)
  assert.equal(planner.getActionsQueueLength(), 0)
  assert.equal(planner.getRequestSlotHandlersCount(), 0)
  assert.equal(planner.getCurInflightRequestsCount(), 0)
  assert.equal(planner.getParallelLimitCorrection(), 0)
  assert.equal(planner.getLastRequestDelay(), 0)
  assert.ok(planner.checks >= 3)
})

test('Independent EventEmitter error path and wrapFetch preserve identity', async t => {
  const runtime = createRuntime(t)
  const events = new EventEmitter<FetchPlannerEventMap>()
  const error = new Error('offline transport failure')
  const planner = new AcceptancePlanner(() => Promise.reject(error), { eventHandler: events })
  const observed: string[] = []
  events.on('request', (_ev, instance) => { assert.equal(instance, planner); observed.push('request') })
  events.on('fetch-error', (ev, instance) => { assert.equal(instance, planner); assert.equal(ev.error, error); observed.push('fetch-error') })
  const failure = assert.rejects(planner.getFetch()('http://offline.test/error'), e => e === error)
  await runtime.advance(0)
  await failure
  assert.deepEqual(observed, ['request', 'fetch-error'])
  assert.equal(planner.getCurInflightRequestsCount(), 0)
  assert.ok(planner.checks > 0)
  const response = new Response('wrapped')
  const fetch = wrapFetch(() => Promise.resolve(response), { jitter: 0 })
  const pending = fetch('http://offline.test/wrap')
  await runtime.advance(0)
  assert.equal(await pending, response)
  assert.equal(response.bodyUsed, false)
  assert.equal(await response.text(), 'wrapped')
})
```
